echo W5 open the app notification settings screen (Show notifications switch)
shell cat /proc/uptime
shell pidof com.mazawallet
shell cmd appops get com.mazawallet POST_NOTIFICATION
shell am start -W -a android.settings.APP_NOTIFICATION_SETTINGS --es android.provider.extra.APP_PACKAGE com.mazawallet --ei android.provider.extra.APP_UID 10068
sleep 3
shell dumpsys activity activities | grep -e mResumedActivity
shell uiautomator dump /sdcard/ui.xml
shell cat /sdcard/ui.xml
shell rm -f /sdcard/ui.xml
